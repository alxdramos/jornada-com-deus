/** Seleciona a passagem do dia com offset para não coincidir com o versículo */
export function getPassagemDoDia(): PassagemDiaria {
  const now = new Date();
  const start = new Date(now.getFullYear(), 0, 1);
  const dayOfYear = Math.floor((now.getTime() - start.getTime()) / 86400000);
  // Offset de 15 para não repetir padrão do versículo
  return PASSAGENS_DIARIAS[(dayOfYear + 15) % PASSAGENS_DIARIAS.length];
}
