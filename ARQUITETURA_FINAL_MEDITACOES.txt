  7. ESTRUTURA DE DADOS
================================================================================

Interface MeditationCard:

export interface MeditationCard {
  id: string;              // "medit-001"
  title: string;           // "Meditação 1"
  duration: string;        // "12 min"
  category: string;        // "MENTE" | "CORPO" | "ESPÍRITO"
  plus: boolean;           // Requer Plus?
  description?: string;    // Primeiras 200 chars
  tags: string[];          // ["PAZ", "DORMIR"]
  image?: string;          // URL imagem
  audioUrl?: string;       // URL R2
}

Mapeamento de Colunas (Excel → TypeScript):

  titulo           →  title
  descricao        →  description (substring 200)
  audioUrl         →  audioUrl (validar HTTP 200)
  duracao_min      →  duration (`${min} min`)
  categoria        →  category (mente→MENTE, corpo→CORPO, etc)
  tags             →  tags (split(","), uppercase)
  imagem_url       →  image (Unsplash default se vazio)
